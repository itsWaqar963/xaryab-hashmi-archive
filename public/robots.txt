import { MetadataRoute } from 'next'

export default function sitemap(): MetadataRoute.Sitemap {
  // Domain deployment ke baad zaroorat ke mutabiq update kar sakte hain
  const baseUrl = 'https://xaryab-hashmi-archive.vercel.app' 

  return [
    {
      url: baseUrl,
      lastModified: new Date(),
      changeFrequency: 'daily',
      priority: 1,
    },
    {
      url: `${baseUrl}/admin`,
      lastModified: new Date(),
      changeFrequency: 'monthly',
      priority: 0.3,
    },
  ]
}
